# 【流向一后端核销调用示例】（以 Python 为例）
import requests
from flask import Flask, request, jsonify

app = Flask(__name__)

@app.route('/api/user/login-by-wechat', methods=['POST'])
def login_by_wechat():
    user_code = request.json.get('code')

    # 请求中枢平台进行核销
    verify_resp = requests.post('https://中枢域名/api/code/verify-universal', json={
        'project': 'your_project',
        'key': 'sec_your_key',
        'code': user_code
    })
    res_data = verify_resp.json()

    if res_data.get('code') == 0 and res_data['data']['valid']:
        wx_openid = res_data['data']['openid']
        # 核销成功，拿着 openid 处理业务系统自身的用户登录或注册
        return jsonify({'status': 'success', 'openid': wx_openid})
    else:
        return jsonify({'status': 'fail', 'msg': res_data.get('message', '验证码无效')}), 400

// 【流向二标准调用示例】
async function handleWechatScanLogin() {
  // 步骤 1: 申请 6 位验证码（传入项目标识与安全密钥）
  const createRes = await fetch('https://中枢域名/api/code/create', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ project: 'your_project', key: 'sec_your_key', session_tag: 'sess_abc123', expire_seconds: 180 })
  });
  const { data } = await createRes.json();
  const code = data.code;
  const pollToken = data.poll_token; // 轮询密钥：业务后端凭此查询状态

  // 网页展示该验证码，引导用户发给公众号
  document.getElementById('code-view').innerText = code;

  // 步骤 2: 定时器轮询状态 (每 1.5 秒查一次，携带 poll_token)
  const timer = setInterval(async () => {
    const checkRes = await fetch(`http://中枢域名/api/code/check?poll_token=${pollToken}`);
    const { data: result } = await checkRes.json();

    if (result && result.verified) {
      clearInterval(timer);
      console.log('用户验证成功，微信 OpenID 为:', result.openid);
      // 业务系统使用此 openid 执行自身登录跳转或账号关联
      window.location.href = '/dashboard';
    }
  }, 1500);
}
【核心架构认知】本平台扮演什么角色？
验证码中枢路由

本系统是一个通用的微信身份中枢与验证码转发路由。它本身不存储外部业务系统的业务逻辑与用户数据，仅负责完成微信用户 ⇄ 6位验证码的撮合校验：
• 外部业务系统向中枢申请验证码或发起核销；
• 校验通过后，中枢把微信唯一标识 openid 交付给外部系统；
• 实际项目拿到 openid 后，根据自身业务需求决定是完成登录、注册、激活还是数据绑定。
• 单向通信原则：本中枢不需要知晓外部网站的具体网址，也不需要向外部网站发送 Webhook 回调，外部项目即使在本地 localhost 也能直接调接口。

验证码默认寿命：3 分钟 (180秒) 单次消费：核销成功后立即销毁防重放 防刷机制：未过期前重复索取返回相同码
